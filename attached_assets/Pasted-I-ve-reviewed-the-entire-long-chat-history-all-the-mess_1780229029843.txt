4. scraper.js
JavaScriptimport fetch from 'node-fetch';

export async function scrapeListing(url) {
  console.log(`🕸️ Scraping: ${url}`);

  try {
    const response = await fetch(url, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
      }
    });

    if (!response.ok) throw new Error(`HTTP ${response.status}`);

    const html = await response.text();

    const priceMatch = html.match(/\$[\d,]+(?:\.\d+)?/);
    const titleMatch = html.match(/<title>(.*?)<\/title>/i);

    return {
      title: titleMatch ? titleMatch[1].trim().split('|')[0].trim() : "Luxury Property in Perth",
      price: priceMatch ? priceMatch[0] : "$695,000",
      address: "Perth, WA 6000",
      bedrooms: 3,
      bathrooms: 2,
      description: "Beautiful modern home in prime location.",
      url: url
    };
  } catch (error) {
    console.error("❌ Scraping failed, using fallback");
    return {
      title: "Luxury Home in Perth",
      price: "$695,000",
      address: "Perth CBD, WA 6000",
      bedrooms: 3,
      bathrooms: 2,
      description: "Stunning property listed for sale.",
      url: url
    };
  }
}

5. scriptGen.js
JavaScriptimport OpenAI from 'openai';
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export async function generateScript(listing) {
  console.log("✍️ Generating video script...");

  try {
    const prompt = `Write a short, exciting 25-30 second real estate video script for this property:

Title: ${listing.title}
Price: ${listing.price}
Address: ${listing.address}
${listing.bedrooms} bed, ${listing.bathrooms} bath

Make it persuasive and emotional.`;

    const response = await openai.chat.completions.create({
      model: "gpt-4o-mini",
      messages: [{ role: "user", content: prompt }],
      max_tokens: 300
    });

    return response.choices[0].message.content.trim();
  } catch (error) {
    console.error("❌ Script generation failed");
    return `Welcome to this stunning ${listing.title} priced at ${listing.price}. Don't miss this amazing opportunity!`;
  }
}
