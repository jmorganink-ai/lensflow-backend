import { defineSchema, defineTable } from "convex/server"; import { v } from "convex/values"; export default defineSchema({ users: defineTable({ tokenIdentifier: v.string(), name: v.optional(v.string()), email: v.optional(v.string()), }).index("by_token", ["tokenIdentifier"]), waitlist: defineTable({ name: v.string(), email: v.string(), }).index("by_email", ["email"]), properties: defineTable({ userId: v.id("users"), address: v.string(), suburb: v.string(), state: v.string(), propertyType: v.union( v.literal("house"), v.literal("apartment"), v.literal("townhouse"), v.literal("land"), v.literal("commercial") ), bedrooms: v.optional(v.number()), bathrooms: v.optional(v.number()), price: v.optional(v.string()), status: v.union( v.literal("active"), v.literal("pending"), v.literal("sold"), v.literal("off_market") ), imageUrl: v.optional(v.string()), notes: v.optional(v.string()), }) .index("by_user", ["userId"]) .index("by_user_and_status", ["userId", "status"]), videos: defineTable({ userId: v.id("users"), propertyId: v.id("properties"), title: v.string(), avatar: v.union(v.literal("mia"), v.literal("oliver"), v.literal("aria"), v.literal("marcus")), status: v.union( v.literal("script"), v.literal("recording"), v.literal("editing"), v.literal("ready"), v.literal("delivered") ), script: v.optional(v.string()), videoUrl: v.optional(v.string()), deliveredToClientId: v.optional(v.id("clients")), deliveredAt: v.optional(v.string()), }) .index("by_user", ["userId"]) .index("by_property", ["propertyId"]) .index("by_user_and_status", ["userId", "status"]), clients: defineTable({ userId: v.id("users"), name: v.string(), email: v.optional(v.string()), phone: v.optional(v.string()), company: v.optional(v.string()), notes: v.optional(v.string()), propertyIds: v.optional(v.array(v.id("properties"))), }).index("by_user", ["userId"]), activities: defineTable({ userId: v.id("users"), type: v.union( v.literal("client_added"), v.literal("client_updated"), v.literal("video_delivered"), v.literal("property_linked") ), description: v.string(), clientId: v.optional(v.id("clients")), videoId: v.optional(v.id("videos")), propertyId: v.optional(v.id("properties")), }).index("by_user", ["userId"]), });
